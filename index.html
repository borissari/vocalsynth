let vids = {}; let keysDown = {}; let allPressed = false; let prevAllPressed = false; let crops = {}; // random crop regions let screenX, screenY, screenW, screenH; const FACE_PARTS = { z: { x: -80, y: -50 }, // eye b: { x: 20, y: -50 }, // eye m: { x: -20, y: 0 }, // nose '.': { x: -40, y: 60 }, // mouth c: { x: 0, y: 0 } // ears placeholder }; function preload() { vids.z = createVideo('vocalsynth_ethan.mp4'); vids.b = createVideo('vocalsynth_dad.mp4'); vids.m = createVideo('vocalsynth_monsang.mp4'); vids['.'] = createVideo('vocalsynth_mom.mp4'); } function setup() { createCanvas(windowWidth, windowHeight); screenW = width * 0.33; screenH = height - 40; screenX = width / 2 - screenW / 2; screenY = 20; for (let k in vids) { vids[k].hide(); vids[k].loop(); vids[k].volume(0); } } function draw() { background(15); // screen frame noFill(); stroke(80); rect(screenX, screenY, screenW, screenH); allPressed = keysDown['z'] && keysDown['b'] && keysDown['m'] && keysDown['.'] && keysDown['c']; // detect transition → generate random crops ONCE if (allPressed && !prevAllPressed) { generateRandomCrops(); } prevAllPressed = allPressed; if (!allPressed) { drawSpread(); } else { drawFace(); } } /* ---------------- SPREAD ---------------- */ function drawSpread() { let spacing = screenW / 5; let i = 0; for (let k of ['z', 'b', 'm', '.', 'c']) { let x = screenX + spacing * i + spacing / 2; let y = screenY + screenH / 2; if (vids[k]) { image(vids[k], x - 80, y - 60, 160, 120); } else { fill(70); rect(x - 60, y - 60, 120, 120); fill(255); textAlign(CENTER, CENTER); text('C', x, y); } i++; } } /* ---------------- FACE ---------------- */ function drawFace() { let cx = screenX + screenW / 2; let cy = screenY + screenH / 2; for (let k in FACE_PARTS) { if (vids[k]) { let part = FACE_PARTS[k]; let crop = crops[k]; drawCropped( vids[k], cx + part.x, cy + part.y, 80, 60, crop ); } } // ears placeholder fill(90); ellipse(cx - 130, cy - 10, 40, 60); ellipse(cx + 130, cy - 10, 40, 60); } /* ---------------- CROPPING ---------------- */ function drawCropped(vid, x, y, w, h, crop) { image( vid, x, y, w, h, crop.sx, crop.sy, crop.sw, crop.sh ); } /* ---------------- RANDOM CROP ---------------- */ function generateRandomCrops() { for (let k in vids) { let vw = vids[k].width; let vh = vids[k].height; let sw = vw * 0.35; let sh = vh * 0.35; let sx = random(0, vw - sw); let sy = random(0, vh - sh); crops[k] = { sx, sy, sw, sh }; } } /* ---------------- INPUT ---------------- */ function keyPressed() { keysDown[key] = true; if (vids[key]) vids[key].volume(1); } function keyReleased() { keysDown[key] = false; if (vids[key]) vids[key].volume(0); }
